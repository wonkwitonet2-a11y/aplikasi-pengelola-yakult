import { useState, useEffect, useRef } from "react";
import {
  TrendingUp,
  Save,
  CheckCircle,
  XCircle,
  FileText,
  PieChart as PieIcon,
  ChevronRight,
  TrendingDown
} from "lucide-react";
import { Transaction, YLTarget, MotivasiConfig } from "../types";

const formatRp = (val: number) => {
  return "Rp" + Math.trunc(val).toLocaleString("id-ID");
};

interface YLViewProps {
  ylName: string;
  onLogout: () => void;
  transactions: Transaction[];
  targetYL: any[];
  onSaveTransaction: (data: Transaction) => Promise<boolean>;
  onSaveTargetYL: (data: any) => Promise<boolean>;
  motivasiConfig: MotivasiConfig;
}

export function YLView({
  ylName,
  onLogout,
  transactions,
  targetYL,
  onSaveTransaction,
  onSaveTargetYL,
  motivasiConfig
}: YLViewProps) {
  const [activeTab, setActiveTab] = useState<"input" | "ringkasan">("input");
  const [selectedDate, setSelectedDate] = useState<string>(
    new Date().toISOString().split("T")[0]
  );

  // Form states
  const [totYo, setTotYo] = useState<number>(0);
  const [totOm, setTotOm] = useState<number>(0);
  const [totOs, setTotOs] = useState<number>(0);
  const [totYt, setTotYt] = useState<number>(0);

  // Sectors rmh, psr, skh, ktr, tk, ib
  const [sectors, setSectors] = useState<Record<string, Record<string, number>>>({
    rmh: { yo: 0, om: 0, os: 0, yt: 0 },
    psr: { yo: 0, om: 0, os: 0, yt: 0 },
    skh: { yo: 0, om: 0, os: 0, yt: 0 },
    ktr: { yo: 0, om: 0, os: 0, yt: 0 },
    tk: { yo: 0, om: 0, os: 0, yt: 0 },
    ib: { yo: 0, om: 0, os: 0, yt: 0 }
  });

  // BB states
  const [bbYo, setBbYo] = useState<number>(0);
  const [bbOm, setBbOm] = useState<number>(0);
  const [bbOs, setBbOs] = useState<number>(0);
  const [bbYt, setBbYt] = useState<number>(0);

  // Kunjungan states
  const [fPlg, setFPlg] = useState<number>(0);
  const [fRk, setFRk] = useState<number>(0);
  const [fRa, setFRa] = useState<number>(0);
  const [fRb, setFRb] = useState<number>(0);

  // PB & Sampah states
  const [pbP, setPbP] = useState<number>(0);
  const [pbS, setPbS] = useState<number>(0);
  const [apkPlg, setApkPlg] = useState<number>(0);
  const [apkBotol, setApkBotol] = useState<number>(0);

  // Divisor E6 target states
  const [pembagiE6, setPembagiE6] = useState<number>(12);
  const [targetVal, setTargetVal] = useState<number>(33);
  const [blnLaluVal, setBlnLaluVal] = useState<number>(30);
  const [thnLaluVal, setThnLaluVal] = useState<number>(28);

  const [saving, setSaving] = useState<boolean>(false);
  const [successMsg, setSuccessMsg] = useState<string>("");

  // Motivasi sliding slideshow state
  const [activeMotivasi, setActiveMotivasi] = useState<string>("Semangat menjalani hari ini dengan tulus!");
  const motivasiIndexRef = useRef<number>(0);

  // Rotate motivasi
  useEffect(() => {
    if (!motivasiConfig.enabled || motivasiConfig.terpilih.length === 0) return;
    const list = motivasiConfig.terpilih;
    setActiveMotivasi(list[0]);

    const timer = setInterval(() => {
      motivasiIndexRef.current = (motivasiIndexRef.current + 1) % list.length;
      setActiveMotivasi(list[motivasiIndexRef.current]);
    }, (motivasiConfig.intervalDetik || 30) * 1000);

    return () => clearInterval(timer);
  }, [motivasiConfig]);

  // Load existing transaction for chosen date
  useEffect(() => {
    const existing = transactions.find(t => t.tanggal === selectedDate);
    if (existing) {
      setTotYo(existing.tot_yo || 0);
      setTotOm(existing.tot_om || 0);
      setTotOs(existing.tot_os || 0);
      setTotYt(existing.tot_yt || 0);

      const secCopy = { ...sectors };
      ["rmh", "psr", "skh", "ktr", "tk", "ib"].forEach(sec => {
        ["yo", "om", "os", "yt"].forEach(prod => {
          secCopy[sec][prod] = (existing as any)[`${sec}_${prod}`] || 0;
        });
      });
      setSectors(secCopy);

      setBbYo(existing.bb_yo || 0);
      setBbOm(existing.bb_om || 0);
      setBbOs(existing.bb_os || 0);
      setBbYt(existing.bb_yt || 0);

      setFPlg(existing.f_plg || 0);
      setFRk(existing.f_rk || 0);
      setFRa(existing.f_ra || 0);
      setFRb(existing.f_rb || 0);

      setPbP(existing.pb_p || 0);
      setPbS(existing.pb_s || 0);
      setApkPlg(existing.apk_plg || 0);
      setApkBotol(existing.apk_botol || 0);
    } else {
      // Clear form
      setTotYo(0); setTotOm(0); setTotOs(0); setTotYt(0);
      setSectors({
        rmh: { yo: 0, om: 0, os: 0, yt: 0 },
        psr: { yo: 0, om: 0, os: 0, yt: 0 },
        skh: { yo: 0, om: 0, os: 0, yt: 0 },
        ktr: { yo: 0, om: 0, os: 0, yt: 0 },
        tk: { yo: 0, om: 0, os: 0, yt: 0 },
        ib: { yo: 0, om: 0, os: 0, yt: 0 }
      });
      setBbYo(0); setBbOm(0); setBbOs(0); setBbYt(0);
      setFPlg(0); setFRk(0); setFRa(0); setFRb(0);
      setPbP(0); setPbS(0); setApkPlg(0); setApkBotol(0);
    }

    // Load matching target configuration
    const currentMonth = selectedDate.substring(0, 7);
    const area = ylName.substring(0, 3);
    const tgtObj = targetYL.find(t => t.bulan === currentMonth);
    if (tgtObj) {
      setTargetVal(tgtObj.target || 33);
      setBlnLaluVal(tgtObj.bln_lalu || 30);
      setThnLaluVal(tgtObj.thn_lalu || 28);
      setPembagiE6(tgtObj.e6 || 12);
    } else {
      setTargetVal(33);
      setBlnLaluVal(30);
      setThnLaluVal(28);
      setPembagiE6(12);
    }
  }, [selectedDate, transactions, targetYL]);

  // Sector calculations
  const getSectorTotal = (prod: "yo" | "om" | "os" | "yt") => {
    return ["rmh", "psr", "skh", "ktr", "tk", "ib"].reduce(
      (sum, sec) => sum + (sectors[sec][prod] || 0),
      0
    );
  };

  const getSectorsGrandTotal = () => {
    return ["yo", "om", "os", "yt"].reduce(
      (sum, p) => sum + getSectorTotal(p as any),
      0
    );
  };

  const isMatched = (prod: "yo" | "om" | "os" | "yt", mainTotal: number) => {
    return getSectorTotal(prod) === mainTotal;
  };

  const isAllSectorsMatched = () => {
    return (
      isMatched("yo", totYo) &&
      isMatched("om", totOm) &&
      isMatched("os", totOs) &&
      isMatched("yt", totYt)
    );
  };

  // Change sector cell value
  const handleSectorChange = (sec: string, prod: string, val: number) => {
    setSectors(prev => ({
      ...prev,
      [sec]: {
        ...prev[sec],
        [prod]: val
      }
    }));
  };

  // Save targets E6
  const handleSaveTargets = async () => {
    const currentMonth = selectedDate.substring(0, 7);
    const success = await onSaveTargetYL({
      nama: ylName,
      bulan: currentMonth,
      target: targetVal,
      bln_lalu: blnLaluVal,
      thn_lalu: thnLaluVal,
      e6: pembagiE6
    });
    if (success) {
      setSuccessMsg("Target & pembagi rata2 berhasil diperbarui!");
      setTimeout(() => setSuccessMsg(""), 2000);
    }
  };

  // Save report
  const handleSaveReport = async () => {
    if (!isAllSectorsMatched()) {
      alert("Pecahan sektor tidak cocok dengan total penjualan utama harian!");
      return;
    }

    setSaving(true);
    const payload: Transaction = {
      tanggal: selectedDate,
      nama: ylName,
      tot_yo: totYo,
      tot_om: totOm,
      tot_os: totOs,
      tot_yt: totYt,
      rmh_yo: sectors.rmh.yo, rmh_om: sectors.rmh.om, rmh_os: sectors.rmh.os, rmh_yt: sectors.rmh.yt,
      psr_yo: sectors.psr.yo, psr_om: sectors.psr.om, psr_os: sectors.psr.os, psr_yt: sectors.psr.yt,
      skh_yo: sectors.skh.yo, skh_om: sectors.skh.om, skh_os: sectors.skh.os, skh_yt: sectors.skh.yt,
      ktr_yo: sectors.ktr.yo, ktr_om: sectors.ktr.om, ktr_os: sectors.ktr.os, ktr_yt: sectors.ktr.yt,
      tk_yo: sectors.tk.yo, tk_om: sectors.tk.om, tk_os: sectors.tk.os, tk_yt: sectors.tk.yt,
      ib_yo: sectors.ib.yo, ib_om: sectors.ib.om, ib_os: sectors.ib.os, ib_yt: sectors.ib.yt,
      bb_yo: bbYo, bb_om: bbOm, bb_os: bbOs, bb_yt: bbYt,
      pb_p: pbP, pb_s: pbS,
      f_plg: fPlg, f_rk: fRk, f_ra: fRa, f_rb: fRb,
      apk_plg: apkPlg, apk_botol: apkBotol
    };

    const success = await onSaveTransaction(payload);
    setSaving(false);
    if (success) {
      setSuccessMsg("Laporan harian berhasil disimpan ke Google Sheet!");
      setTimeout(() => setSuccessMsg(""), 2500);
    }
  };

  // Setoran calculations
  const setYo = totYo * 2040;
  const setOm = totOm * 2130;
  const setOs = totOs * 2130;
  const setYt = totYt * 2460;
  const grandSetoran = setYo + setOm + setOs + setYt;

  // Monthly stats calculations for RINGKASAN
  const currentMonth = selectedDate.substring(0, 7);
  const currentMonthTxs = transactions.filter(t => t.tanggal.startsWith(currentMonth));
  
  let mYo = 0, mOm = 0, mOs = 0, mYt = 0;
  let mRmh = 0, mPsr = 0, mSkh = 0, mKtr = 0, mTk = 0, mIb = 0;
  let mRk = 0, mPlg = 0, mRa = 0, mRb = 0;
  let mPb = 0, mBb = 0, mSampahBotol = 0;
  let mKompensasiHarian = 0;

  currentMonthTxs.forEach(t => {
    mYo += t.tot_yo || 0;
    mOm += t.tot_om || 0;
    mOs += t.tot_os || 0;
    mYt += t.tot_yt || 0;

    mRmh += (t.rmh_yo||0) + (t.rmh_om||0) + (t.rmh_os||0) + (t.rmh_yt||0);
    mPsr += (t.psr_yo||0) + (t.psr_om||0) + (t.psr_os||0) + (t.psr_yt||0);
    mSkh += (t.skh_yo||0) + (t.skh_om||0) + (t.skh_os||0) + (t.skh_yt||0);
    mKtr += (t.ktr_yo||0) + (t.ktr_om||0) + (t.ktr_os||0) + (t.ktr_yt||0);
    mTk += (t.tk_yo||0) + (t.tk_om||0) + (t.tk_os||0) + (t.tk_yt||0);
    mIb += (t.ib_yo||0) + (t.ib_om||0) + (t.ib_os||0) + (t.ib_yt||0);

    mRk += t.f_rk || 0;
    mPlg += t.f_plg || 0;
    mRa += t.f_ra || 0;
    mRb += t.f_rb || 0;

    mPb += (t.pb_p || 0) + (t.pb_s || 0);
    mBb += (t.bb_yo || 0) + (t.bb_om || 0) + (t.bb_os || 0) + (t.bb_yt || 0);
    mSampahBotol += t.apk_botol || 0;

    // Daily kompensasi
    const kYo = Math.max(0, (t.tot_yo||0) - (t.tk_yo || 0)) * 60;
    const kOm = Math.max(0, (t.tot_om||0) - (t.tk_om || 0)) * 70;
    const kOs = Math.max(0, (t.tot_os||0) - (t.tk_os || 0)) * 70;
    const kYt = Math.max(0, (t.tot_yt||0) - (t.tk_yt || 0)) * 140;
    mKompensasiHarian += (kYo + kOm + kOs + kYt);
  });

  const mTotalSales = mYo + mOm + mOs + mYt;
  // Use E6 pembagi as the divisor for ALL harian rata-rata!
  const mRata2 = mTotalSales / pembagiE6;

  // Kompensasi monthly factor calculation
  const getKompensasiFactor = (avg: number) => {
    if (avg < 200) return 338;
    if (avg >= 200 && avg < 250) return 374;
    if (avg >= 250 && avg < 280) return 409;
    if (avg >= 280 && avg < 300) return 417;
    if (avg >= 300 && avg < 330) return 424;
    if (avg >= 330 && avg < 350) return 428;
    return 432; // avg >= 350
  };

  const kompFactor = getKompensasiFactor(mRata2);
  const mKompensasiBulanan = mTotalSales * kompFactor;

  const pctOfTotal = (val: number) => {
    return mTotalSales > 0 ? Math.trunc((val / mTotalSales) * 100) : 0;
  };

  const getPctString = (num: number, denom: number) => {
    return denom > 0 ? Math.trunc((num / denom) * 100) + "%" : "0%";
  };

  return (
    <div className="min-h-screen bg-slate-50 pb-20">
      {/* Header Area */}
      <div className="bg-gradient-to-r from-red-600 to-rose-600 p-4 pb-6 text-white rounded-b-[2rem] shadow-lg relative shrink-0">
        <button
          onClick={onLogout}
          className="absolute top-4 right-4 bg-black/20 hover:bg-black/30 border border-white/20 text-xs font-bold px-3 py-1.5 rounded-xl transition-all"
        >
          Keluar
        </button>
        <div className="text-center mt-2">
          <h2 className="text-xs font-black tracking-widest uppercase">Modul Lapangan Yakult</h2>
          <h3 className="text-lg font-black mt-1">{ylName}</h3>
          <p className="text-[10px] text-red-100 font-bold bg-red-800/40 px-3 py-0.5 rounded-full inline-block mt-1">Jember 1</p>
        </div>
      </div>

      {/* Floating Rotating Motivasi Banner */}
      {motivasiConfig.enabled && (
        <div className="mx-4 -mt-3 relative z-10 bg-gradient-to-r from-amber-50 to-orange-50 border-2 border-amber-300 p-3 rounded-2xl shadow-sm text-center">
          <p className="text-xs font-black text-amber-900 italic transition-all duration-300">
            ✨ {activeMotivasi}
          </p>
        </div>
      )}

      {/* Main Content Area */}
      <main className="p-4 space-y-4 max-w-lg mx-auto">
        {/* Tab Navigation */}
        <div className="flex gap-1.5 bg-slate-200 p-1 rounded-xl">
          <button
            onClick={() => setActiveTab("input")}
            className={`flex-1 py-2.5 text-xs font-black rounded-lg transition-all ${activeTab === "input" ? "bg-white text-rose-600 shadow" : "text-slate-600"}`}
          >
            INPUT DATA
          </button>
          <button
            onClick={() => setActiveTab("ringkasan")}
            className={`flex-1 py-2.5 text-xs font-black rounded-lg transition-all ${activeTab === "ringkasan" ? "bg-white text-rose-600 shadow" : "text-slate-600"}`}
          >
            RINGKASAN
          </button>
        </div>

        {/* INPUT DATA TAB */}
        {activeTab === "input" && (
          <div className="space-y-4">
            {/* 1. Date selector */}
            <div className="bg-white rounded-2xl p-4 border border-slate-100 shadow-sm space-y-3">
              <div>
                <label className="text-[9px] font-black text-slate-400 uppercase tracking-wider block mb-1">Tanggal Transaksi</label>
                <input
                  type="date"
                  value={selectedDate}
                  onChange={(e) => setSelectedDate(e.target.value)}
                  className="w-full p-2 bg-slate-50 border border-slate-200 rounded-xl font-bold text-xs text-slate-800 outline-none"
                />
              </div>
            </div>

            {/* 2. Penjualan Utama & Setoran */}
            <div className="bg-white rounded-2xl p-4 border border-slate-100 shadow-sm space-y-4">
              <h2 className="text-xs font-black text-slate-800 uppercase tracking-wider border-l-4 border-emerald-500 pl-2">
                1. Total Jual & Setoran Harian
              </h2>
              <div className="space-y-3">
                {/* Headers */}
                <div className="grid grid-cols-12 gap-2 text-[9px] font-bold text-slate-400 uppercase">
                  <div className="col-span-3">Produk</div>
                  <div className="col-span-4 text-center">Jual (Btl)</div>
                  <div className="col-span-5 text-right">Setoran</div>
                </div>

                {/* YO Row */}
                <div className="grid grid-cols-12 gap-2 items-center">
                  <span className="col-span-3 font-bold text-red-600 text-xs">YO</span>
                  <input
                    type="number"
                    value={totYo}
                    onChange={(e) => setTotYo(Math.max(0, Number(e.target.value)))}
                    className="col-span-4 p-1.5 bg-slate-50 border border-slate-200 rounded-lg text-center font-bold text-xs"
                  />
                  <span className="col-span-5 text-right font-mono font-bold text-xs text-slate-600">{formatRp(setYo)}</span>
                </div>

                {/* OM Row */}
                <div className="grid grid-cols-12 gap-2 items-center">
                  <span className="col-span-3 font-bold text-amber-500 text-xs">OM</span>
                  <input
                    type="number"
                    value={totOm}
                    onChange={(e) => setTotOm(Math.max(0, Number(e.target.value)))}
                    className="col-span-4 p-1.5 bg-slate-50 border border-slate-200 rounded-lg text-center font-bold text-xs"
                  />
                  <span className="col-span-5 text-right font-mono font-bold text-xs text-slate-600">{formatRp(setOm)}</span>
                </div>

                {/* OS Row */}
                <div className="grid grid-cols-12 gap-2 items-center">
                  <span className="col-span-3 font-bold text-pink-500 text-xs">OS</span>
                  <input
                    type="number"
                    value={totOs}
                    onChange={(e) => setTotOs(Math.max(0, Number(e.target.value)))}
                    className="col-span-4 p-1.5 bg-slate-50 border border-slate-200 rounded-lg text-center font-bold text-xs"
                  />
                  <span className="col-span-5 text-right font-mono font-bold text-xs text-slate-600">{formatRp(setOs)}</span>
                </div>

                {/* YT Row */}
                <div className="grid grid-cols-12 gap-2 items-center">
                  <span className="col-span-3 font-bold text-blue-600 text-xs">YT</span>
                  <input
                    type="number"
                    value={totYt}
                    onChange={(e) => setTotYt(Math.max(0, Number(e.target.value)))}
                    className="col-span-4 p-1.5 bg-slate-50 border border-slate-200 rounded-lg text-center font-bold text-xs"
                  />
                  <span className="col-span-5 text-right font-mono font-bold text-xs text-slate-600">{formatRp(setYt)}</span>
                </div>

                <hr className="border-dashed border-slate-100" />
                <div className="flex justify-between items-center text-xs font-black text-slate-800">
                  <span>TOTAL ESTIMASI SETORAN</span>
                  <span className="text-sm font-mono text-emerald-600">{formatRp(grandSetoran)}</span>
                </div>
              </div>
            </div>

            {/* 3. Sektor breakdown */}
            <div className="bg-white rounded-2xl p-4 border border-slate-100 shadow-sm space-y-4">
              <h2 className="text-xs font-black text-slate-800 uppercase tracking-wider border-l-4 border-blue-500 pl-2">
                2. Sektor Distribusi (Pecah Botol)
              </h2>
              <p className="text-[9px] text-slate-400 font-bold uppercase leading-relaxed">
                Pecah total botol terjual hari ini ke sektor-sektor berikut. Total harus COCOK dengan bagian penjualan di atas!
              </p>

              {["rmh", "psr", "skh", "ktr", "tk", "ib"].map((secKey) => {
                const label = secKey === "rmh" ? "Rumah" : secKey === "psr" ? "Pasar" : secKey === "skh" ? "Sekolah" : secKey === "ktr" ? "Kantor" : secKey === "tk" ? "Toko" : "IB";
                return (
                  <div key={secKey} className="bg-slate-50/50 p-3 rounded-xl border border-slate-100 space-y-2">
                    <span className="text-[10px] font-black text-slate-700 uppercase tracking-wider block">{label}</span>
                    <div className="grid grid-cols-4 gap-1.5">
                      {["yo", "om", "os", "yt"].map((prod) => {
                        const color = prod === "yo" ? "text-red-600" : prod === "om" ? "text-amber-500" : prod === "os" ? "text-pink-500" : "text-blue-600";
                        return (
                          <div key={prod} className="text-center space-y-1">
                            <span className={`text-[9px] font-black uppercase ${color}`}>{prod.toUpperCase()}</span>
                            <input
                              type="number"
                              value={sectors[secKey][prod]}
                              onChange={(e) => handleSectorChange(secKey, prod, Math.max(0, Number(e.target.value)))}
                              className="w-full p-1 bg-white border border-slate-200 rounded text-center text-xs font-bold"
                            />
                          </div>
                        );
                      })}
                    </div>
                  </div>
                );
              })}

              {/* Sektor Match Validation Block */}
              <div className={`p-3.5 rounded-2xl border-2 space-y-2 ${isAllSectorsMatched() ? "bg-emerald-50/70 border-emerald-300" : "bg-rose-50/70 border-rose-200"}`}>
                <div className="flex items-center justify-between text-xs font-black">
                  <span>Pencocokan Sektor Jual</span>
                  <span className={isAllSectorsMatched() ? "text-emerald-700 font-extrabold flex items-center gap-0.5" : "text-rose-600 font-extrabold"}>
                    {isAllSectorsMatched() ? "✓ COCOK" : "❌ BELUM COCOK"}
                  </span>
                </div>
                <div className="grid grid-cols-2 gap-x-4 gap-y-1.5 text-[10px] font-bold text-slate-500">
                  <div className="flex justify-between items-center">
                    <span>YO: {getSectorTotal("yo")} / {totYo}</span>
                    <span className={isMatched("yo", totYo) ? "text-emerald-600" : "text-rose-600"}>
                      {isMatched("yo", totYo) ? "Pas" : "Mis"}
                    </span>
                  </div>
                  <div className="flex justify-between items-center">
                    <span>OM: {getSectorTotal("om")} / {totOm}</span>
                    <span className={isMatched("om", totOm) ? "text-emerald-600" : "text-rose-600"}>
                      {isMatched("om", totOm) ? "Pas" : "Mis"}
                    </span>
                  </div>
                  <div className="flex justify-between items-center">
                    <span>OS: {getSectorTotal("os")} / {totOs}</span>
                    <span className={isMatched("os", totOs) ? "text-emerald-600" : "text-rose-600"}>
                      {isMatched("os", totOs) ? "Pas" : "Mis"}
                    </span>
                  </div>
                  <div className="flex justify-between items-center">
                    <span>YT: {getSectorTotal("yt")} / {totYt}</span>
                    <span className={isMatched("yt", totYt) ? "text-emerald-600" : "text-rose-600"}>
                      {isMatched("yt", totYt) ? "Pas" : "Mis"}
                    </span>
                  </div>
                </div>
              </div>
            </div>

            {/* Kunjungan & Propaganda */}
            <div className="bg-white rounded-2xl p-4 border border-slate-100 shadow-sm space-y-4">
              <h2 className="text-xs font-black text-slate-800 uppercase tracking-wider border-l-4 border-slate-500 pl-2">
                3. Kunjungan, PB, & Sampah Botol
              </h2>
              {/* Kunjungan Grid */}
              <div className="grid grid-cols-4 gap-2 text-center">
                <div>
                  <label className="text-[8px] font-black text-slate-400 uppercase block mb-1">PLG (Tgt)</label>
                  <input
                    type="number"
                    value={fPlg}
                    onChange={(e) => setFPlg(Math.max(0, Number(e.target.value)))}
                    className="w-full p-1.5 bg-slate-50 border border-slate-200 rounded-lg text-center text-xs font-bold"
                  />
                </div>
                <div>
                  <label className="text-[8px] font-black text-slate-400 uppercase block mb-1">RK (Kunj)</label>
                  <input
                    type="number"
                    value={fRk}
                    onChange={(e) => setFRk(Math.max(0, Number(e.target.value)))}
                    className="w-full p-1.5 bg-slate-50 border border-slate-200 rounded-lg text-center text-xs font-bold"
                  />
                </div>
                <div>
                  <label className="text-[8px] font-black text-slate-400 uppercase block mb-1">RA (Aktif)</label>
                  <input
                    type="number"
                    value={fRa}
                    onChange={(e) => setFRa(Math.max(0, Number(e.target.value)))}
                    className="w-full p-1.5 bg-slate-50 border border-slate-200 rounded-lg text-center text-xs font-bold"
                  />
                </div>
                <div>
                  <label className="text-[8px] font-black text-slate-400 uppercase block mb-1">RB (Beli)</label>
                  <input
                    type="number"
                    value={fRb}
                    onChange={(e) => setFRb(Math.max(0, Number(e.target.value)))}
                    className="w-full p-1.5 bg-slate-50 border border-slate-200 rounded-lg text-center text-xs font-bold"
                  />
                </div>
              </div>

              {/* PB & Sampah Grid */}
              <div className="grid grid-cols-2 gap-3">
                <div className="p-3 bg-slate-50 rounded-xl border border-slate-100">
                  <span className="text-[9px] font-black text-slate-400 uppercase block mb-2">Propaganda Baru (PB)</span>
                  <div className="grid grid-cols-2 gap-2">
                    <div className="text-center">
                      <span className="text-[8px] font-bold text-slate-400 block mb-0.5">Pagi</span>
                      <input
                        type="number"
                        value={pbP}
                        onChange={(e) => setPbP(Math.max(0, Number(e.target.value)))}
                        className="w-full p-1 bg-white border border-slate-200 rounded text-center text-xs font-bold"
                      />
                    </div>
                    <div className="text-center">
                      <span className="text-[8px] font-bold text-slate-400 block mb-0.5">Sore</span>
                      <input
                        type="number"
                        value={pbS}
                        onChange={(e) => setPbS(Math.max(0, Number(e.target.value)))}
                        className="w-full p-1 bg-white border border-slate-200 rounded text-center text-xs font-bold"
                      />
                    </div>
                  </div>
                </div>

                <div className="p-3 bg-slate-50 rounded-xl border border-slate-100">
                  <span className="text-[9px] font-black text-slate-400 uppercase block mb-2">Apk Sampah Botol</span>
                  <div className="grid grid-cols-2 gap-2">
                    <div className="text-center">
                      <span className="text-[8px] font-bold text-slate-400 block mb-0.5">Plg</span>
                      <input
                        type="number"
                        value={apkPlg}
                        onChange={(e) => setApkPlg(Math.max(0, Number(e.target.value)))}
                        className="w-full p-1 bg-white border border-slate-200 rounded text-center text-xs font-bold"
                      />
                    </div>
                    <div className="text-center">
                      <span className="text-[8px] font-bold text-slate-400 block mb-0.5">Botol</span>
                      <input
                        type="number"
                        value={apkBotol}
                        onChange={(e) => setApkBotol(Math.max(0, Number(e.target.value)))}
                        className="w-full p-1 bg-white border border-slate-200 rounded text-center text-xs font-bold"
                      />
                    </div>
                  </div>
                </div>
              </div>
            </div>

            {/* Balik Botol */}
            <div className="bg-white rounded-2xl p-4 border border-slate-100 shadow-sm space-y-4">
              <h2 className="text-xs font-black text-slate-800 uppercase tracking-wider border-l-4 border-rose-500 pl-2">
                4. Balik Botol (BB)
              </h2>
              <div className="grid grid-cols-4 gap-2 text-center">
                <div>
                  <span className="text-[9px] font-black text-red-600 block mb-1">YO</span>
                  <input
                    type="number"
                    value={bbYo}
                    onChange={(e) => setBbYo(Math.max(0, Number(e.target.value)))}
                    className="w-full p-1.5 bg-slate-50 border border-slate-200 rounded-lg text-center text-xs font-bold"
                  />
                </div>
                <div>
                  <span className="text-[9px] font-black text-amber-500 block mb-1">OM</span>
                  <input
                    type="number"
                    value={bbOm}
                    onChange={(e) => setBbOm(Math.max(0, Number(e.target.value)))}
                    className="w-full p-1.5 bg-slate-50 border border-slate-200 rounded-lg text-center text-xs font-bold"
                  />
                </div>
                <div>
                  <span className="text-[9px] font-black text-pink-500 block mb-1">OS</span>
                  <input
                    type="number"
                    value={bbOs}
                    onChange={(e) => setBbOs(Math.max(0, Number(e.target.value)))}
                    className="w-full p-1.5 bg-slate-50 border border-slate-200 rounded-lg text-center text-xs font-bold"
                  />
                </div>
                <div>
                  <span className="text-[9px] font-black text-blue-600 block mb-1">YT</span>
                  <input
                    type="number"
                    value={bbYt}
                    onChange={(e) => setBbYt(Math.max(0, Number(e.target.value)))}
                    className="w-full p-1.5 bg-slate-50 border border-slate-200 rounded-lg text-center text-xs font-bold"
                  />
                </div>
              </div>

              {successMsg && (
                <p className="text-[11px] font-black text-emerald-600 text-center animate-pulse">{successMsg}</p>
              )}

              <button
                onClick={handleSaveReport}
                disabled={saving || !isAllSectorsMatched()}
                className="w-full bg-red-600 text-white hover:bg-red-700 font-extrabold text-xs py-3 rounded-xl transition-all shadow-md flex items-center justify-center gap-1.5 disabled:opacity-40"
              >
                <Save className="w-4 h-4" />
                {saving ? "MENYIMPAN..." : "SIMPAN LAPORAN HARIAN"}
              </button>
            </div>
          </div>
        )}

        {/* RINGKASAN TAB */}
        {activeTab === "ringkasan" && (
          <div className="space-y-4">
            {/* Target Settings Card */}
            <div className="bg-white rounded-2xl p-4 border border-slate-100 shadow-sm space-y-3">
              <h2 className="text-xs font-black text-slate-800 uppercase tracking-wider border-l-4 border-slate-500 pl-2">
                Sasaran & Target Bulanan
              </h2>
              <p className="text-[9px] text-slate-400 font-bold uppercase">
                Atur target rata-rata dan nilai E6 pembagi untuk menghitung rasio secara akurat.
              </p>
              <div className="grid grid-cols-4 gap-2 text-center text-[10px] font-bold">
                <div>
                  <span className="text-slate-400 block mb-1">Target</span>
                  <input
                    type="number"
                    value={targetVal}
                    onChange={(e) => setTargetVal(Number(e.target.value))}
                    className="w-full p-1.5 bg-slate-50 border border-slate-200 rounded text-center text-xs font-bold"
                  />
                </div>
                <div>
                  <span className="text-slate-400 block mb-1">Bln Lalu (Q4)</span>
                  <input
                    type="number"
                    value={blnLaluVal}
                    onChange={(e) => setBlnLaluVal(Number(e.target.value))}
                    className="w-full p-1.5 bg-slate-50 border border-slate-200 rounded text-center text-xs font-bold"
                  />
                </div>
                <div>
                  <span className="text-slate-400 block mb-1">Thn Lalu</span>
                  <input
                    type="number"
                    value={thnLaluVal}
                    onChange={(e) => setThnLaluVal(Number(e.target.value))}
                    className="w-full p-1.5 bg-slate-50 border border-slate-200 rounded text-center text-xs font-bold"
                  />
                </div>
                <div>
                  <span className="text-slate-400 block mb-1">E6 Pembagi</span>
                  <input
                    type="number"
                    value={pembagiE6}
                    onChange={(e) => setPembagiE6(Number(e.target.value))}
                    className="w-full p-1.5 bg-slate-50 border border-slate-200 rounded text-center text-xs font-bold"
                  />
                </div>
              </div>
              <button
                onClick={handleSaveTargets}
                className="w-full bg-slate-900 text-white text-xs font-bold py-2 rounded-xl hover:bg-slate-950 transition-all mt-2"
              >
                Simpan Konfigurasi Target
              </button>
            </div>

            {/* Akumulasi Penjualan */}
            <div className="bg-white rounded-2xl p-4 border border-slate-100 shadow-sm space-y-4">
              <h2 className="text-xs font-black text-slate-800 uppercase tracking-wider border-l-4 border-red-600 pl-2">
                Akumulasi Penjualan (Bulan Ini)
              </h2>
              <div className="space-y-3.5">
                {/* YO progress */}
                <div className="space-y-1">
                  <div className="flex justify-between text-[11px] font-bold text-slate-700">
                    <span className="text-red-600">YO Original</span>
                    <span>{mYo} btl ({pctOfTotal(mYo)}%)</span>
                  </div>
                  <div className="w-full h-2 bg-slate-100 rounded-full overflow-hidden">
                    <div className="h-full bg-red-600" style={{ width: `${pctOfTotal(mYo)}%` }} />
                  </div>
                </div>

                {/* OM progress */}
                <div className="space-y-1">
                  <div className="flex justify-between text-[11px] font-bold text-slate-700">
                    <span className="text-amber-500">OM Mango</span>
                    <span>{mOm} btl ({pctOfTotal(mOm)}%)</span>
                  </div>
                  <div className="w-full h-2 bg-slate-100 rounded-full overflow-hidden">
                    <div className="h-full bg-amber-500" style={{ width: `${pctOfTotal(mOm)}%` }} />
                  </div>
                </div>

                {/* OS progress */}
                <div className="space-y-1">
                  <div className="flex justify-between text-[11px] font-bold text-slate-700">
                    <span className="text-pink-500">OS Stroberi</span>
                    <span>{mOs} btl ({pctOfTotal(mOs)}%)</span>
                  </div>
                  <div className="w-full h-2 bg-slate-100 rounded-full overflow-hidden">
                    <div className="h-full bg-pink-500" style={{ width: `${pctOfTotal(mOs)}%` }} />
                  </div>
                </div>

                {/* YT progress */}
                <div className="space-y-1">
                  <div className="flex justify-between text-[11px] font-bold text-slate-700">
                    <span className="text-blue-600">YT Light</span>
                    <span>{mYt} btl ({pctOfTotal(mYt)}%)</span>
                  </div>
                  <div className="w-full h-2 bg-slate-100 rounded-full overflow-hidden">
                    <div className="h-full bg-blue-600" style={{ width: `${pctOfTotal(mYt)}%` }} />
                  </div>
                </div>

                <hr className="border-slate-100" />
                <div className="bg-red-50 p-3 rounded-xl border border-red-100 text-center space-y-1">
                  <span className="text-[10px] font-black text-red-950 uppercase tracking-wider block">Grand Total Jual</span>
                  <span className="text-xl font-black text-rose-600 block">{mTotalSales} btl</span>
                  <span className="text-[10px] text-slate-500 font-bold block">Rata-Rata harian: {Math.trunc(mRata2)} btl/hari</span>
                </div>
              </div>
            </div>

            {/* Estimasi Kompensasi */}
            <div className="bg-white rounded-2xl p-4 border border-slate-100 shadow-sm space-y-3">
              <h2 className="text-xs font-black text-slate-800 uppercase tracking-wider border-l-4 border-emerald-500 pl-2">
                Rincian Estimasi Kompensasi
              </h2>
              <div className="space-y-2">
                <div className="bg-slate-50 p-2.5 rounded-xl border border-slate-150 flex justify-between items-center text-xs">
                  <span className="font-bold text-slate-500">1. Kompensasi Harian (Akm)</span>
                  <span className="font-extrabold text-slate-800">{formatRp(mKompensasiHarian)}</span>
                </div>
                <div className="bg-slate-50 p-2.5 rounded-xl border border-slate-150 flex justify-between items-center text-xs">
                  <span className="font-bold text-slate-500">2. Estimasi Kompensasi Bulanan</span>
                  <span className="font-extrabold text-slate-800">{formatRp(mKompensasiBulanan)}</span>
                </div>
                <div className="bg-emerald-50 p-3.5 rounded-2xl border-2 border-emerald-300 text-center space-y-1">
                  <span className="text-[10px] font-black text-emerald-950 uppercase block">Total Take Home Bonus</span>
                  <span className="text-2xl font-black text-emerald-600 block">{formatRp(mKompensasiHarian + mKompensasiBulanan)}</span>
                  <span className="text-[9px] text-slate-400 font-medium block">*Nilai ini bersifat estimasi berdasarkan formula baku dropping Jember 1</span>
                </div>
              </div>
            </div>

            {/* Ratios & Comparisons */}
            <div className="bg-white rounded-2xl p-4 border border-slate-100 shadow-sm space-y-3">
              <h2 className="text-xs font-black text-slate-800 uppercase tracking-wider border-l-4 border-slate-500 pl-2">
                Performa & Rasio Kunjungan
              </h2>
              <div className="grid grid-cols-2 gap-2 text-center text-xs font-bold">
                <div className="bg-slate-50 p-2 rounded-xl border border-slate-100">
                  <span className="text-[8px] text-slate-400 block mb-0.5">RK vs PLG</span>
                  <span className="text-sm font-black text-slate-800">{getPctString(mRk, mPlg)}</span>
                </div>
                <div className="bg-slate-50 p-2 rounded-xl border border-slate-100">
                  <span className="text-[8px] text-slate-400 block mb-0.5">RA vs RK</span>
                  <span className="text-sm font-black text-slate-800">{getPctString(mRa, mRk)}</span>
                </div>
                <div className="bg-slate-50 p-2 rounded-xl border border-slate-100">
                  <span className="text-[8px] text-slate-400 block mb-0.5">RB vs RA</span>
                  <span className="text-sm font-black text-slate-800">{getPctString(mRb, mRa)}</span>
                </div>
                <div className="bg-slate-50 p-2 rounded-xl border border-slate-100">
                  <span className="text-[8px] text-slate-400 block mb-0.5">RB vs PLG</span>
                  <span className="text-sm font-black text-slate-800">{getPctString(mRb, mPlg)}</span>
                </div>
              </div>

              <div className="grid grid-cols-3 gap-2 text-center text-xs font-bold mt-2">
                <div className="bg-red-50/50 p-2 rounded-xl border border-red-100">
                  <span className="text-[8px] text-slate-400 block mb-0.5">vs Target</span>
                  <span className="text-sm font-black text-red-600">{getPctString(mRata2, targetVal)}</span>
                </div>
                <div className="bg-red-50/50 p-2 rounded-xl border border-red-100">
                  <span className="text-[8px] text-slate-400 block mb-0.5">vs Bln Lalu</span>
                  <span className="text-sm font-black text-red-600">{getPctString(mRata2, blnLaluVal)}</span>
                </div>
                <div className="bg-red-50/50 p-2 rounded-xl border border-red-100">
                  <span className="text-[8px] text-slate-400 block mb-0.5">vs Thn Lalu</span>
                  <span className="text-sm font-black text-red-600">{getPctString(mRata2, thnLaluVal)}</span>
                </div>
              </div>
            </div>

            {/* Riwayat Table */}
            <div className="bg-white rounded-2xl border border-slate-100 shadow-sm overflow-hidden">
              <div className="p-3 bg-red-950 text-white font-black text-[10px] uppercase tracking-wider">
                Daftar Riwayat Terakhir Saya
              </div>
              <div className="overflow-x-auto">
                <table className="w-full text-left text-xs border-collapse font-mono">
                  <thead>
                    <tr className="bg-slate-50 border-b border-slate-100 text-slate-400 font-bold uppercase text-[9px]">
                      <th className="p-2.5">Tanggal</th>
                      <th className="p-2.5 text-right">Total</th>
                      <th className="p-2.5 text-right">Breakdown</th>
                    </tr>
                  </thead>
                  <tbody className="divide-y divide-slate-100 font-bold">
                    {currentMonthTxs.slice(-5).reverse().map((t, idx) => {
                      const dayTotal = t.tot_yo + t.tot_om + t.tot_os + t.tot_yt;
                      return (
                        <tr key={idx} className="hover:bg-slate-50 text-slate-800">
                          <td className="p-2.5 truncate max-w-[80px]">{t.tanggal}</td>
                          <td className="p-2.5 text-right text-emerald-600">{dayTotal} btl</td>
                          <td className="p-2.5 text-right text-[10px] text-slate-500 font-medium">
                            {t.tot_yo}/{t.tot_om}/{t.tot_os}/{t.tot_yt}
                          </td>
                        </tr>
                      );
                    })}
                  </tbody>
                </table>
              </div>
            </div>
          </div>
        )}
      </main>
    </div>
  );
}
